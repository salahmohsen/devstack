## Canonical Step 2: New `packages/api/src/client/tanstack-start/orpc.ts`

```typescript
import { createORPCClient, eventIteratorToUnproxiedDataStream } from "@orpc/client";
import { createRouterClient, type RouterClient } from "@orpc/server";
import { OpenAPILink } from "@orpc/openapi-client/fetch";
import { createTanstackQueryUtils } from "@orpc/tanstack-query";
import type { RouterContract, JsonifiedClient } from "@orpc/contract";
import { createIsomorphicFn } from "@tanstack/react-start";
import { getRequestHeaders } from "@tanstack/react-start/server";

import { auth } from "@tsu-stack/auth";
import { type OrpcContextMediaBucket } from "@tsu-stack/core";
import { createDb, runWithDb } from "@tsu-stack/db";
import { ENV_WEB_ISOMORPHIC } from "@tsu-stack/env/web/env.isomorphic";
import { createLogger } from "@tsu-stack/logger/server";
import { createGitHubOctokit } from "@tsu-stack/github/infrastructure/octokit/client";
import { resolvePublicGitHubSource } from "@tsu-stack/github/shared/public-github-source";
import { createAgentWorker, type AgentWorkerEnv } from "#@/lib/agent/worker";

import { type AppRouter, appRouter } from "#@/routers/index";
import contractJson from "@tsu-stack/api/contract.json"; // generated by task-7

const getORPCClient = createIsomorphicFn()
  .server(() =>
    createRouterClient(appRouter, {
      context: async () =>
        runWithDb(createDb(), async () => {
          const headers = getRequestHeaders();
          const session = await auth.api.getSession({ headers });
          const [{ createGitHubOctokit }, { resolvePublicGitHubSource }] = await Promise.all([
            import("@tsu-stack/github/infrastructure/octokit/client"),
            import("@tsu-stack/github/shared/public-github-source")
          ]);
          const githubSource = await resolvePublicGitHubSource();
          const envAny = process.env as unknown as AgentWorkerEnv;
          const agentWorker = envAny.AGENT_WORKER
            ? createAgentWorker(envAny)
            : {
                fetch(): Promise<Response> {
                  throw new Error(
                    "Agent worker is not configured. Collapsed NestJS deploy should set MASTRA in the OrpcContext instead."
                  );
                }
              };

          return {
            github: {
              binding: {},
              privateOctokit: createGitHubOctokit(
                githubSource.isConfigured ? githubSource : undefined
              ),
              publicOctokit: createGitHubOctokit(
                githubSource.isConfigured ? githubSource : undefined
              ),
              source: githubSource
            },
            logger: createLogger({ operation: "web__client__orpc" }),
            session,
            mediaBucket: process.env.MEDIA_BUCKET as unknown as OrpcContextMediaBucket,
            agentWorker
          };
        }),
      interceptors: [
        (options) => {
          const db = createDb();
          return runWithDb(db, () => options.next());
        }
      ]
    })
  )
  .client((): JsonifiedClient<RouterClient<AppRouter>> => {
    // In SSR we'd never reach here; this is the BROWSER bundle.
    const link = new OpenAPILink(contractJson as unknown as RouterContract<typeof appRouter>, {
      url: `${ENV_WEB_ISOMORPHIC.VITE_SERVER_URL}`,
      fetch(url, options) {
        return window.fetch(url, { ...options, credentials: "include" }).catch((err) => {
          console.error("[openapi-fetch-failed]", { url, err });
          throw err;
        });
      }
    });
    return createORPCClient(link) as JsonifiedClient<RouterClient<AppRouter>>;
  });

export const orpcClient = getORPCClient();
export const orpcQueryClient = createTanstackQueryUtils(
  orpcClient as unknown as RouterClient<AppRouter>
);
export { eventIteratorToUnproxiedDataStream };
```

## Pitfall: `JsonifiedClient` is REQUIRED

The OpenAPILink → HTTP serialization normalizes dates, BigInts,
etc. to JSON-friendly equivalents. The raw `RouterClient<AppRouter>`
won't TS-check after this normalization. **Use `JsonifiedClient`**
(or accept the cast and trust runtime).

## Pitfall: Streaming endpoints (`chatStream`, `detectStacks`,

`organizeStacks`)

For `EventIterator`-returning procedures, `OpenAPILink` serializes
chunks to Server-Sent Events on the wire. The browser can:

```typescript
import { eventIteratorToUnproxiedDataStream } from "@orpc/client";

const stream = await orpcClient.agent.detectStacks.streamBotStreamed(
  { projectId },
  { signal: abortController.signal }
);
for await (const chunk of eventIteratorToUnproxiedDataStream(stream)) {
  // ...
}
```

`eventIteratorToUnproxiedDataStream` is re-exported from this file
(line `export { ... }`). Existing TanStack Query stream usages
that worked with `RPCLink` continue working with `OpenAPILink`.

## Pitfall: `contract.json` regeneration timing

`@tsu-stack/api/prebuild` script regenerates `contract.json`
BEFORE `tsc` build. But `apps/web` reads it directly via
`import contractJson from "@tsu-stack/api/contract.json"`. Web's
typecheck (`vp typecheck`) requires the JSON exists at
THAT TIME.

If web CI runs before `pnpm --filter @tsu-stack/api build` (and
thus before `pnpm build:contract`), it will error. Two fixes:

- **A**: Add `pnpm --filter @tsu-stack/api build` (or just
  `pnpm build:contract`) as a dependency of web's `prebuild`.
- **B**: Add `packages/api/contract.json` to Git (defeats purpose of
  minification but does work for stability — pick A).

We use option A.

```json
// packages/web/package.json
"scripts": {
  "prebuild": "pnpm --filter @tsu-stack/api build",
  ...
}
```

With NestJS serving on Server, web also depends on server running.
In dev, run `pnpm dev` (the existing dotenvx workflow) after
`pnpm --filter @tsu-stack/server dev`.

## Pitfall: TanStack Query keys for `OpenAPILink`

`@orpc/tanstack-query`'s `queryOptions` factory reads the contract's
keys via `createORPCQueryOptions`. The keys are derived from the
path of the procedure ('router.procedure'). These match what
`OpenAPILink` expects on the wire. **No change to `useQuery` /
`useMutation` call sites needed.**
